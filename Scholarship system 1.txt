"""Student Scholarship Eligibility System.

Evaluates a student's scholarship eligibility using academic marks,
attendance percentage, and monthly family income.
"""

# Eligibility rules
MIN_MARKS = 80
MIN_ATTENDANCE = 85
MAX_MONTHLY_INCOME = 50000
FULL_SCHOLARSHIP_MARKS = 90


def get_name(prompt):
    """Ask for a name until a non-empty value is entered."""
    while True:
        name = input(prompt).strip()
        if name:
            return name
        print("Name cannot be empty.")


def get_number(prompt, minimum, maximum=None):
    """Ask for a number until it falls inside the allowed range."""
    while True:
        try:
            value = float(input(prompt))
        except ValueError:
            print("Invalid input. Please enter a number.")
            continue

        if value < minimum or (maximum is not None and value > maximum):
            if maximum is None:
                print(f"Value must be at least {minimum}.")
            else:
                print(f"Value must be between {minimum} and {maximum}.")
            continue

        return value


def find_ineligibility_reasons(marks, attendance, income):
    """Check each criterion separately and return a list of failed reasons."""
    reasons = []

    if marks < MIN_MARKS:
        reasons.append(f"Marks are below the required {MIN_MARKS}%.")
    if attendance < MIN_ATTENDANCE:
        reasons.append(f"Attendance is below the required {MIN_ATTENDANCE}%.")
    if income > MAX_MONTHLY_INCOME:
        reasons.append(f"Family income exceeds Rs. {MAX_MONTHLY_INCOME:,}.")

    return reasons


def determine_scholarship(marks):
    """Return the scholarship type for an eligible student."""
    if marks >= FULL_SCHOLARSHIP_MARKS:
        return "Full Scholarship"
    else:
        return "Partial Scholarship"


def print_report(name, marks, attendance, income, reasons):
    """Display the complete scholarship evaluation report."""
    print("\n===== SCHOLARSHIP EVALUATION REPORT =====")
    print(f"Student Name:     {name}")
    print(f"Academic Marks:   {marks:.2f}%")
    print(f"Attendance:       {attendance:.2f}%")
    print(f"Monthly Income:   Rs. {income:,.2f}")
    print("-" * 41)

    if reasons:
        print("Status:           NOT ELIGIBLE")
        print("Reasons:")
        for reason in reasons:
            print(f"  - {reason}")
    else:
        print("Status:           ELIGIBLE")
        print(f"Scholarship:      {determine_scholarship(marks)}")


def main():
    print("===== STUDENT SCHOLARSHIP ELIGIBILITY SYSTEM =====")

    # Collect and validate input
    name = get_name("Enter student name: ")
    marks = get_number("Enter academic marks (0-100): ", 0, 100)
    attendance = get_number("Enter attendance percentage (0-100): ", 0, 100)
    income = get_number("Enter monthly family income (Rs.): ", 0)

    # Evaluate and report
    reasons = find_ineligibility_reasons(marks, attendance, income)
    print_report(name, marks, attendance, income, reasons)


if __name__ == "__main__":
    main()
